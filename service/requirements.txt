# transitive minor release breaking activation at 2am is not a trade worth making for newer versions.

uvicorn[standard]==0.34.3 # ASGI server
cryptography==50.0.0     # Ed25519 signing -- the same library the desktop client verifies with
stripe==11.4.1           # webhook signature verification and subscription state
pydantic==2.13.4         # request validation (arrives with FastAPI; named so it is explicit)

# Dev
pytest==9.1.1
